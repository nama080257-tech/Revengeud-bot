// ================= WAUTH =================
async function generateWauthKey(product, duration, userId) {
  try {
    const apiKey = product === "temp"
      ? CONFIG.WAUTH_TEMP_API_KEY
      : CONFIG.WAUTH_ULTIMATE_API_KEY;

    const res = await axios.post(
      "https://wauth.cc/api/v1/generate",
      {
        duration: duration,
        note: `Discord ID: ${userId}`
      },
      {
        headers: { Authorization: apiKey }
      }
    );

    return res.data.key;
  } catch (err) {
    console.log("Wauth error:", err.message);
    return null;
  }
}

// ================= PAYPAL WEBHOOK =================
app.post("/paypal-webhook", async (req, res) => {
  try {
    const { user_id, product, duration } = req.body;

    const key = await generateWauthKey(product, duration, user_id);
    if (!key) return res.sendStatus(500);

    const guild = await client.guilds.fetch(CONFIG.DISCORD_GUILD_ID);
    const member = await guild.members.fetch(user_id);
    await member.roles.add(CONFIG.CUSTOMER_ROLE_ID);

    const db = loadDB();
    if (!db.users[user_id]) db.users[user_id] = {};
    db.users[user_id][product] = key;
    saveDB(db);

    const user = await client.users.fetch(user_id);

    const embed = new EmbedBuilder()
      .setTitle("✅ Purchase Successful")
      .setDescription("Your access has been activated successfully.")
      .addFields(
        { name: "🔑 Key", value: key },
        { name: "📊 Status", value: "Active" },
        { name: "🛒 Product", value: product }
      );

    await user.send({ embeds: [embed] });

    res.sendStatus(200);

  } catch (err) {
    console.log(err);
    res.sendStatus(500);
  }
});
